cmake_minimum_required(VERSION 3.20.0)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(OSPREY_SYNNAX_FULL_CLIENT "Construct the complete Synnax client (experimental)" OFF)
option(OSPREY_SYNNAX_AUTH_ONLY "Build the barebones plaintext Synnax login client" OFF)
if(OSPREY_SYNNAX_FULL_CLIENT AND OSPREY_SYNNAX_AUTH_ONLY)
    message(FATAL_ERROR "Select either the complete client or the login-only client")
endif()
if(NOT DEFINED BOARD)
    set(BOARD nucleo_h723zg)
endif()
if(OSPREY_SYNNAX_FULL_CLIENT OR OSPREY_SYNNAX_AUTH_ONLY)
    list(APPEND EXTRA_CONF_FILE "${CMAKE_CURRENT_SOURCE_DIR}/full-client.conf")
endif()

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(osprey-app LANGUAGES C CXX)

include(port/runtime.cmake)
include(cmake/dtcm.cmake)
include(cmake/synnax.cmake)
target_sources(app PRIVATE src/main.cpp src/atomic64.cpp port/std_thread.cpp)
target_link_libraries(app PRIVATE synnax_client)
